        private async void btnStart_Click(object sender, EventArgs e)

            // await ProcessTickerAsync("EURUSD", "5min", startDate, endDate);
            // Fire and forget - don't await so the click handler returns immediately
            _ = RunMonitoringLoopAsync(tickers, candlePeriod, speed, timeFrame, startDate, _monitoringCts.Token, atrMultiplier);
        }

        private async Task RunMonitoringLoopAsync(
            List<string> tickers
            , int candlePeriod
            , string speed
            , string timeFrame
            , DateTime startDate
            , CancellationToken cancellationToken
            , decimal atrMultiplier
            )
        {
            int intervalMs = GetIntervalMs(speed);





            while (!cancellationToken.IsCancellationRequested)
            {
                try
                {
                    // Process all tickers concurrently
                    var tasks = tickers.Select(ticker => ProcessTickerAsync(ticker, 0, candlePeriod, timeFrame, startDate, atrMultiplier));
                    await Task.WhenAll(tasks);

                    startDate = GetDate(timeFrame, startDate);
                    // Wait for the specified interval
                    await Task.Delay(intervalMs, cancellationToken);
                }
                catch (OperationCanceledException)
                {
                    // Monitoring was cancelled
                    break;
                }
                catch (Exception ex)
                {
                    System.Diagnostics.Debug.WriteLine($"Error in monitoring loop: {ex.Message}");
                }
            }
        }

        private int GetIntervalMs(string speed)
        {
            return speed switch
            {
                "1x" => 1000,
                "2x" => 500,
                "5x" => 200,
                "10x" => 100,
                _ => 1000, // Default to 1x if unknown
            };
        }

        private DateTime GetDate(string timeFrame, DateTime date)
        {
            return timeFrame switch
            {
                "1min" => date.AddMinutes(1),
                "5min" => date.AddMinutes(5),
                "15min" => date.AddMinutes(15),
                "30min" => date.AddMinutes(30),
                "1h" => date.AddHours(1),
                "4h" => date.AddHours(4),
                "1d" => date.AddDays(1),
                _ => date.AddDays(5),
            };
        }

        private void StopMonitoring()
        {
            _isMonitoring = false;
            _monitoringCts?.Cancel();
            _monitoringCts?.Dispose();
            _monitoringCts = null;
            candles.Clear();
            btnStart.Text = "Start";
        }

        private async Task ProcessTickerAsync(string ticker, int index, int candlePeriod, string timeFrame, DateTime startDate, decimal atrMultiplier)
        {
            try
            {
                var candlesFromService = new List<Candle>();
                if (chkIsBacktesting.Checked && candles.Count == 0)
                {

                    candlesFromService = await _fmpService.GetIntradayCandlesAsync(ticker, timeFrame, startDate.AddDays(-candlePeriod).ToString("yyyy-MM-dd"), startDate.ToString("yyyy-MM-dd"));

                    // var test = candles.Where(t=>t.Low == 1.33395m && t.High == 1.33479m);

                    await _fmp.GetCandlesAsync(ticker, timeFrame, candlePeriod, startDate);
                }

                



                var estStartDate = startDate.ConvertLocalToEstTime();

                candles = candlesFromService.Where(c => c.Date <= estStartDate)
                   .OrderBy(c => c.Date).ToList(); // Ensure candles are sorted by DateTime


                if (candles == null || !candles.Any())
                    return;

                lblCandleTime.Text = $"@Candle: {candles.FirstOrDefault()?.Date.ConvertEstToLocalTime().ToString("yyyy-MM-dd HH:mm")}";

                //var watcher = new DisplacementWatcher(_alert, ticker, timeFrame);
                //var result = watcher.Detect(candles, index, candlePeriod, atrMultiplier);
                index = candles.Count - 1; // Set index to the last candle for detection
                CandleAnalyzer candleAnalyzer = new CandleAnalyzer(candles, candlePeriod);
                var result = candleAnalyzer.Detect(index, atrMultiplier, ticker, timeFrame);

                if (result.IsDisplacement)
                {
                    // Ensure thread-safety when updating UI
                    if (InvokeRequired)
                    {
                        Invoke(() => OnDisplacementDetected(result.Ticker, result.TimeFrame, result.DateTime, result.Direction));
                    }
                    else
                    {
                        OnDisplacementDetected(result.Ticker, result.TimeFrame, result.DateTime, result.Direction);
                    }
                }
            }
            catch (Exception ex)
            {
                // Log or handle errors per ticker
                System.Diagnostics.Debug.WriteLine($"Error processing {ticker}: {ex.Message}");
            }
        }

        private void gridAlert_CellClick(object? sender, DataGridViewCellEventArgs e)
        {
            if (e.RowIndex < 0)
                return;

            if (e.ColumnIndex < 0)
                return;

            if (gridAlert.Columns[e.ColumnIndex].Name == "StopButton")
            {
                var alert = _activeAlerts[e.RowIndex];

                _activeAlerts.Remove(alert);

                if (_activeAlerts.Count == 0)
                {
                    _alert.StopAlarm();
                }
            }
        }

        private void OnDisplacementDetected(string ticker, string timeframe, DateTime time, string direction)
        {
            bool alreadyExists = _activeAlerts.Any(x =>
                x.Ticker == ticker &&
                x.TimeFrame == timeframe);

            //if (alreadyExists)
            //    return;

            _activeAlerts.Add(new ActiveAlert
            {
                Ticker = ticker,
                TimeFrame = timeframe,
                AlertTime = time,
                Direction = direction
            });

            _alert.StartAlarm();
        }

        private void SaveDate()
        {
            var settings = new Data
            {
                Tickers = txtTickers.Text.Split(";", StringSplitOptions.RemoveEmptyEntries)
                .Distinct()
                .ToList(),
                TimeFrame = cboTimeFrame.SelectedItem?.ToString() ?? "5m"
            };

            var json = JsonSerializer.Serialize(settings, new JsonSerializerOptions
            {
                WriteIndented = true
            });

            File.WriteAllText(SettingsFile, json);

        }

        private void LoadData()
        {
            if (!File.Exists(SettingsFile))
                return;

            var json = File.ReadAllText(SettingsFile);

            var settings = JsonSerializer.Deserialize<Data>(json);

            txtTickers.Text = string.Join(";", settings?.Tickers ?? Array.Empty<string>());
            cboTimeFrame.SelectedItem = settings?.TimeFrame ?? "5m";
        }

        private async Task ScanTickerAsync(string ticker)
        {
            // Stronger setup:
            var signal4h15m = await _engine.CheckFromFmpAsync(_fmp, ticker, "4hour", "15min");
            if (signal4h15m.ShouldAlert)
            {
                AddAlertToGrid(signal4h15m);
                _alert.StartAlarm();
                return;
            }

            // Faster setup:
            var signal1h5m = await _engine.CheckFromFmpAsync(_fmp, ticker, "1hour", "5min");
            if (signal1h5m.ShouldAlert)
            {
                AddAlertToGrid(signal1h5m);
                _alert.StartAlarm();
            }
        }

        private void AddAlertToGrid(Trading.Smc.v2.SmcAlertSignal signal)
        {
            if (InvokeRequired)
            {
                Invoke(() => AddAlertToGrid(signal));
                return;
            }
            bool alreadyExists = _activeAlerts.Any(x =>
                x.Ticker == signal.Ticker &&
                x.TimeFrame == signal.LowerTimeFrame);
            if (alreadyExists)
                return;
            _activeAlerts.Add(new ActiveAlert
            {
                Ticker = signal.Ticker,
                TimeFrame = signal.LowerTimeFrame,
               // AlertTime = signal.,
                Direction = signal.Direction.ToString()
            });
        }
